/* ---------- Утилиты дат (строки yyyy-MM-dd, Москва) ---------- */
function today_() { return Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd'); }
function d_(s) { return new Date(s + 'T00:00:00Z'); }
function fmt_(d) { return d.toISOString().slice(0, 10); }
function isWork_(s) { const g = d_(s).getUTCDay(); return g !== 0 && g !== 6; }
function addWork_(s, n) { const d = d_(s); while (n > 0) { d.setUTCDate(d.getUTCDate() + 1); const g = d.getUTCDay(); if (g !== 0 && g !== 6) n--; } return fmt_(d); }
function workBetween_(a, b) { let n = 0; const d = d_(a), e = d_(b); while (d < e) { d.setUTCDate(d.getUTCDate() + 1); const g = d.getUTCDay(); if (g !== 0 && g !== 6) n++; } return n; }
function ru_(s) { return s ? s.slice(8, 10) + '.' + s.slice(5, 7) : ''; }
function dmy_(s, t) {
  const m = String(s || '').match(/(\d{1,2})\.(\d{1,2})(?:\.(\d{2,4}))?/);
  if (!m) return '';
  const tt = t || today_();
  let y = m[3] ? Number(m[3].length === 2 ? '20' + m[3] : m[3]) : Number(tt.slice(0, 4));
  const pad = n => String(n).padStart(2, '0');
  let r = y + '-' + pad(m[2]) + '-' + pad(m[1]);
  if (!m[3] && (d_(r) - d_(tt)) / 864e5 > 30) r = (y - 1) + r.slice(4);
  return r;
}

/* ---------- Доменная логика ---------- */
function platform_(url) { if (!url) return ''; for (const [re, n] of PLATFORMS) if (re.test(url)) return n; return 'Карьерный сайт'; }
function tgNick_(s) {
  s = String(s || '');
  const m = s.match(/(?:^|[^\w.@])@([A-Za-z][A-Za-z0-9_]{3,31})(?![\w.@])/) || s.match(/t\.me\/([A-Za-z][A-Za-z0-9_]{3,31})/i);
  return m ? m[1] : '';
}

/** Автостатус контакта. due = какой пинг пора отправить. */
function contactState_(c, t) {
  if (c.stage === 'отказ') return { state: 'отказ' };
  if (c.stage === 'игнор') return { state: 'игнор' };
  if (STAGE_RANK[c.stage] >= 2) return { state: c.stage };
  if (c.replied || c.stage === 'ответ/переписка') return { state: 'ответ/переписка' };
  if (!c.p1) return { state: 'написать', due: (c.imported || c.step) ? null : 'p1' };
  if (c.imported && !c.tg && !c.p3) return { state: 'ждём ответ' };
  if (c.p3) return workBetween_(c.p3, t) >= 1 ? { state: 'игнор', auto: true } : { state: 'ждём ответ' };
  if (c.p2) return workBetween_(c.p2, t) >= 1 ? { state: 'пинг 3', due: 'p3' } : { state: 'ждём ответ' };
  return workBetween_(c.p1, t) >= 1 ? { state: 'пинг 2', due: 'p2' } : { state: 'ждём ответ' };
}
function isAnswered_(c) { return !!(c.replied || STAGE_RANK[c.stage] >= 1); }
function fill_(text, comp, c, extra) {
  comp = comp || {};
  return String(text)
    .replace(/\{площадка\}/g, PLATFORM_PREP[comp.platform] || 'вашем карьерном сайте')
    .replace(/\{ссылка\}/g, comp.url || '')
    .replace(/\{компания\}/g, comp.name || '')
    .replace(/\{дата\}/g, ru_((extra && extra.date) || (c && c.stepAt) || ''));
}

/** Разбор курсовой таблицы (значения как отображаются). Чистая функция — тестируется отдельно. */
function parseCourse_(rows, t) {
  const companies = [], contacts = [];
  let cur = null;
  for (let i = 1; i < rows.length; i++) {
    const r = Array.from({ length: 14 }, (_, j) => String(rows[i][j] == null ? '' : rows[i][j]));
    const [a, b, dt, src, contact, p1, p2, p3, st] = r;
    const comment = (r[13] || '').trim();
    if (/ОДНА КОМПАНИЯ|СТРУКТУРУ ТАБЛИЦЫ/.test(a + contact)) continue;
    const isComp = !!(a.trim() || b.trim());
    if (isComp) {
      const url = (b.trim() || src.trim());
      cur = { id: companies.length + 1, name: a.trim() || (cur ? cur.name : ''), url, platform: platform_(url), date: dmy_(dt, t), comment, rejected: /отказ/i.test(st) ? (dmy_(dt, t) || t) : '', created: t };
      companies.push(cur);
    } else if (!cur) continue;
    if (!isComp && !cur.date && dt.trim()) cur.date = dmy_(dt, t);
    if (!isComp && src.trim() && /^http/.test(src.trim()) && src.trim() !== cur.url && (!cur.url || /vacanc|vakans/i.test(src))) { cur.url = src.trim(); cur.platform = platform_(cur.url); }
    if (contact.trim()) {
      const c = { id: contacts.length + 1, cid: cur.id, company: cur.name, contact: contact.trim(), tg: tgNick_(contact), p1: dmy_(p1, t), p2: dmy_(p2, t), p3: dmy_(p3, t), replied: '', stage: '', stageAt: '', scen: '', step: '', stepAt: '', ignNotified: '', imported: '1', note: isComp ? '' : comment, from: '' };
      const last = c.p3 || c.p2 || c.p1 || cur.date;
      if (!isComp) {
        if (/ответ|переписк/i.test(st)) c.replied = last;
        else if (/отказ/i.test(st)) { c.stage = 'отказ'; c.stageAt = last; c.replied = last; }
        else if (/игнор/i.test(st)) c.stage = 'игнор';
      }
      // уже отработанный игнор не должен заново приходить уведомлением
      if (contactState_(c, t).state === 'игнор') c.ignNotified = t;
      contacts.push(c);
    }
  }
  return { companies, contacts };
}

function poll() {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(2000)) return;
  try {
    const p = props_();
    const r = tg_('getUpdates', { offset: Number(p.getProperty('OFFSET') || 0), timeout: 0, allowed_updates: ['message', 'callback_query'] });
    if (r.ok) r.result.forEach(u => {
      p.setProperty('OFFSET', String(u.update_id + 1));
      try { handle_(u); } catch (e) { console.error(e && e.stack || e); try { send_('⚠️ Ошибка: ' + esc_(e.message)); } catch (_) { } }
    });
    maybeDaily_();
  } finally { lock.releaseLock(); }
}

function maybeDaily_() {

}

function handle_(u) {
  if (u.callback_query) return onCallback_(u.callback_query);
  const m = u.message; if (!m) return;
  const text = m.text || m.caption || '';
  if (!owner_()) {
    if (/^\/start/.test(text)) { props_().setProperty('OWNER_ID', String(m.from.id)); send_('Привет, Анна! Я на связи 👋\n\n' + HELP); }
    return;
  }
  if (String(m.from.id) !== owner_()) return;
  if (m.forward_origin || m.forward_date) return onForward_(m);
  if (/^\/(meme|мем)/i.test(text)) return addMeme_(m);
  if (/^\/(start|help)/i.test(text)) return send_(HELP);
  if (/^\/(today|сегодня)/i.test(text)) return daily_();
  if (/^\/(templates|шаблоны)/i.test(text)) return sendAllTemplates_();
  if (/^\/(cancel|отмена)/i.test(text)) { props_().deleteProperty('AWAIT'); return send_('Ок, отменила.'); }

  if (aw) return onAwait_(JSON.parse(aw), text);
  send_('Перешли мне сообщение рекрутера — прикреплю его к контакту. Всё остальное — в мини-аппе «Отклики».');
}

function doPost(e) {
  let out;
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const req = JSON.parse(e.postData.contents);
    const user = verifyInitData_(req.initData || '');
    if (!user || String(user.id) !== owner_()) out = { error: 'forbidden' };
    else { if (req.action !== 'load') api_(req.action, req.payload || {}); out = load_(); }
  } catch (err) { out = { error: String(err && err.message || err) }; }
  finally { lock.releaseLock(); }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

function verifyInitData_(initData) {
  if (!initData) return null;
  const params = {};
  initData.split('&').forEach(p => { const i = p.indexOf('='); params[decodeURIComponent(p.slice(0, i).replace(/\+/g, ' '))] = decodeURIComponent(p.slice(i + 1).replace(/\+/g, ' ')); });
  const hash = params.hash; delete params.hash;
  const dcs = Object.keys(params).sort().map(k => k + '=' + params[k]).join('\n');
  const secret = Utilities.computeHmacSha256Signature(props_().getProperty('TOKEN'), 'WebAppData');
  const sig = Utilities.computeHmacSignature(Utilities.MacAlgorithm.HMAC_SHA_256, Utilities.newBlob(dcs).getBytes(), secret);
  const hex = sig.map(b => ((b + 256) % 256).toString(16).padStart(2, '0')).join('');
  if (hex !== hash) return null;
  if (Date.now() / 1000 - Number(params.auth_date || 0) > 7 * 86400) return null;
  return JSON.parse(params.user || 'null');
}

function load_() {
  const t = today_();
  const contacts = read_('contacts').map(c => Object.assign(c, contactState_(c, t)));
  return {
    today: t, checkChat: CHECK_CHAT,
    companies: read_('companies'), contacts,
    messages: read_('messages'), events: read_('events'), mood: read_('mood'),
    templates: read_('templates'), steps: STEPS, platformPrep: PLATFORM_PREP,
  };
}

const COMPANY_FIELDS = ['name', 'url', 'platform', 'date', 'comment', 'rejected'];
const CONTACT_FIELDS = ['contact', 'p1', 'p2', 'p3', 'replied', 'stage', 'note', 'step', 'stepAt'];

function api_(action, p) {
  const t = today_();
  if (action === 'addCompany') {
    const comp = { id: nextId_('companies'), name: p.name || '', url: p.url || '', platform: p.platform || platform_(p.url), date: p.date || t, comment: p.comment || '', rejected: '', created: t };
    append_('companies', comp);
    logEvent_({ cid: comp.id }, 'company');
    appendMany_('contacts', (p.contacts || []).filter(s => String(s).trim()).map(s => newContact_(comp, s)));
  } else if (action === 'updateCompany') {
    const comp = byId_('companies', p.id);
    const oldName = comp.name;
    COMPANY_FIELDS.forEach(k => { if (k in p.fields) comp[k] = p.fields[k]; });
    update_('companies', comp);
    if (comp.name !== oldName) read_('contacts').filter(c => c.cid === String(comp.id)).forEach(c => { c.company = comp.name; update_('contacts', c); });
  } else if (action === 'deleteCompany') {
    const s = sheet_('contacts');
    read_('contacts').filter(c => c.cid === String(p.id)).map(c => c._row).sort((a, b) => b - a).forEach(r => s.deleteRow(r));
    const comp = byId_('companies', p.id); if (comp) sheet_('companies').deleteRow(comp._row);
  } else if (action === 'addContact') {
    const comp = byId_('companies', p.cid);
    append_('contacts', newContact_(comp, p.contact));
  } else if (action === 'updateContact') {
    const c = byId_('contacts', p.id);
    const f = p.fields;
    if ('contact' in f) { c.contact = f.contact; c.tg = tgNick_(f.contact); }
    if ('stage' in f && f.stage !== c.stage) { c.stageAt = t; if (f.stage && f.stage !== 'игнор') logEvent_(c, 'stage', f.stage); if (f.stage && f.stage !== 'игнор' && !c.replied && !('replied' in f)) { c.replied = t; logEvent_(c, 'reply'); } }
    if ('replied' in f && f.replied && !c.replied) logEvent_(c, 'reply');
    CONTACT_FIELDS.forEach(k => { if (k in f && k !== 'contact') c[k] = f[k]; });
    update_('contacts', c);
  } else if (action === 'deleteContact') {
    const c = byId_('contacts', p.id); if (c) sheet_('contacts').deleteRow(c._row);
  } else if (action === 'markSent') {
    markSent_(p.id, p.kind);
  } else if (action === 'scenario') {
    onScenario_(p.id, p.code, p.extra);
  } else if (action === 'sendTemplate') {
    const c = byId_('contacts', p.id), comp = byId_('companies', c.cid);
    sendTpl_(c, comp, [p.code], (read_('templates').find(x => x.code === p.code) || {}).title || p.code);
  } else if (action === 'saveTemplate') {
    const tp = read_('templates').find(x => x.code === p.code); tp.text = p.text; update_('templates', tp);
  } else throw new Error('unknown action ' + action);
}

function newContact_(comp, s) {
  return { id: nextId_('contacts'), cid: comp.id, company: comp.name, contact: String(s).trim(), tg: tgNick_(s), p1: '', p2: '', p3: '', replied: '', stage: '', stageAt: '', scen: '', step: '', stepAt: '', ignNotified: '', imported: '', note: '', from: '' };
}
